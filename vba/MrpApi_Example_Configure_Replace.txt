' 1) EXAMPLE_JWT sabitini BOS birakin (artik settings.json kullanilacak):
Private Const EXAMPLE_JWT As String = ""

' 2) MrpApi_Example_Configure prosedürünü su sekilde degistirin:

Public Sub MrpApi_Example_Configure()
    If MrpApiDesktopSettings_Configure() Then
        Debug.Print "MrpApi_Configure: Desktop Teklif settings.json"
        Debug.Print "  path: " & MrpApiDesktopSettings_Path()
        Debug.Print "  baseUrl: " & MrpApiDesktopSettings_BaseUrl()
        Debug.Print "  jwt len: " & CStr(Len(MrpApiDesktopSettings_Jwt()))
    Else
        MsgBox "Desktop Teklif JWT bulunamadi." & vbCrLf & _
               MrpApiDesktopSettings_Path() & vbCrLf & vbCrLf & _
               "Once Desktop Teklif > Ayarlar'dan JWT kaydedin.", vbExclamation
    End If
End Sub
